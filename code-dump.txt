/**
 * Public entry point.
 *
 * Overlay: renders Console.js directly in this document.
 * Embedded: renders console-frame.html inside the supplied mount point.
 */

function resolveMountPoint(value) {
    const fallback = "#devtools";

    function resolve(value) {
        if (value instanceof Element) return value;

        if (typeof value !== "string" || !value.trim()) {
            return null;
        }

        try {
            return document.querySelector(value);
        } catch {
            return null;
        }
    }

    const mount = resolve(value) ?? resolve(fallback);

    if (!mount) {
        throw new TypeError(
            'No valid mount point found. Provide an element or create a "#devtools" element.'
        );
    }

    return mount;
}

function resolveHeight(value) {
    const fallback = "60dvh";

    if (
        typeof value === "string" &&
        value.trim() &&
        CSS.supports("height", value.trim())
    ) {
        return value.trim();
    }

    return fallback;
}

/**
 * Stable public controller. The active renderer can change without
 * changing the object returned by DevTools.mount().
 */
export function createConsoleHost(options = {}) {
    let mode = options.mode ?? "embedded";

    if (mode !== "embedded" && mode !== "overlay") {
        throw new TypeError('mode must be "embedded" or "overlay".');
    }

    const settings = {
        ...options,
        mode,
        mountPoint: resolveMountPoint(options.mountPoint),
        height: resolveHeight(options.height),
        captureConsole: options.captureConsole ?? true,
        mirrorToNative: options.mirrorToNative ?? mode === "embedded",
        showCopyButtons: options.showCopyButtons ?? false,
        allowDragOutsideBounds: options.allowDragOutsideBounds ?? true,
        logTestOutput: options.logTestOutput ?? true
    };

    let active = null;
    let destroyed = false;
    let switching = false;

    const suites = new Map();

    function mountCurrent() {
        const mountOptions = {
            ...settings,
            mode,
            onModeToggle: switchMode
        };

        active =
            mode === "embedded"
                ? createEmbeddedConsole(mountOptions)
                : createOverlayConsole(mountOptions);

        // Restore project test suites after a renderer change.
        for (const definition of suites.values()) {
            active.registerSuite(definition);
        }
    }

    function switchMode() {
        if (destroyed || switching) return;

        switching = true;

        try {
            const nextMode = mode === "embedded" ? "overlay" : "embedded";

            active?.destroy();
            active = null;
            mode = nextMode;
            settings.mode = nextMode;

            mountCurrent();
        } catch (error) {
            console.error("[DevTools] Could not switch modes:", error);
        } finally {
            switching = false;
        }
    }

    mountCurrent();

    return {
        get mode() {
            return mode;
        },

        get host() {
            return active?.host ?? null;
        },

        get frame() {
            return active?.frame ?? null;
        },

        get console() {
            return active?.console ?? active?.devConsole ?? null;
        },

        get devConsole() {
            return active?.devConsole ?? active?.console ?? null;
        },

        get tests() {
            return this.devConsole?.tests ?? null;
        },

        registerSuite(definition) {
            if (destroyed) {
                throw new Error("Developer console has been destroyed.");
            }

            if (suites.has(definition.id)) {
                throw new Error(
                    `Suite "${definition.id}" is already registered.`
                );
            }

            const unregisterActive = active.registerSuite(definition);
            suites.set(definition.id, definition);

            return () => {
                suites.delete(definition.id);
                return unregisterActive();
            };
        },

        unregisterSuite(id) {
            suites.delete(id);
            return active?.unregisterSuite(id) ?? false;
        },

        getSuites() {
            return active.getSuites();
        },

        destroy() {
            if (destroyed) return;
            destroyed = true;

            active?.destroy();
            active = null;
            suites.clear();
        }
    };
}

function createOverlayConsole({
    onModeToggle,
    captureConsole = true,
    mirrorToNative = false,
    showCopyButtons = false,
    allowDragOutsideBounds = true,
    logTestOutput = true
} = {}) {
    ensureConsoleStyles();

    const consoleElement = document.createElement("div");
    consoleElement.className = "mobile-dev-console-overlay";
    document.body.appendChild(consoleElement);

    const previousDevConsole = window.devConsole;

    let consoleInstance;
    let inspectorHost;

    const dispatchInspectorEvent = (event, payload = {}) => {
        window.dispatchEvent(
            new CustomEvent("devtools-inspect-event", {
                detail: { event, payload }
            })
        );
    };

    try {
        // Overlay mode runs in the application document, so inspection can
        // call the host directly instead of requiring the iframe message bridge.
        inspectorHost = new DOMInspectorHost({
            document,
            mountElement: consoleElement,
            onPick({ id, details }) {
                dispatchInspectorEvent("picked", { id, details });
            },
            onPickCancelled() {
                dispatchInspectorEvent("pick-cancelled");
            },
            onSelectionChange({ id, details }) {
                dispatchInspectorEvent("selection-changed", { id, details });
            }
        });

        consoleInstance = new Console({
            element: consoleElement,
            mode: "overlay",
            captureConsole,
            mirrorToNative,
            showCopyButtons,
            allowDragOutsideBounds,
            logTestOutput,
            inspectionBridge: inspectorHost
        });
    } catch (error) {
        inspectorHost?.destroy();
        consoleElement.remove();
        throw error;
    }

    const onModeToggleRequest = () => onModeToggle?.();

    consoleElement.addEventListener(
        "console-mode-toggle-request",
        onModeToggleRequest
    );

    let destroyed = false;

    const api = {
        mode: "overlay",
        host: consoleElement,
        frame: null,

        console: consoleInstance,
        devConsole: consoleInstance,

        tests: consoleInstance.tests,

        registerSuite(definition) {
            return consoleInstance.tests.registerSuite(definition);
        },

        unregisterSuite(id) {
            return consoleInstance.tests.unregisterSuite(id);
        },

        getSuites() {
            return consoleInstance.tests.getSuites();
        },

        destroy() {
            if (destroyed) return;
            destroyed = true;

            consoleElement.removeEventListener(
                "console-mode-toggle-request",
                onModeToggleRequest
            );

            consoleInstance.destroy();
            inspectorHost?.destroy();
            inspectorHost = null;
            consoleElement.remove();

            if (window.devConsole === consoleInstance) {
                if (previousDevConsole == null) {
                    delete window.devConsole;
                } else {
                    window.devConsole = previousDevConsole;
                }
            }
        }
    };

    return api;
}

function serialize(value) {
    if (value instanceof Error) {
        return {
            name: value.name,
            message: value.message,
            stack: value.stack
        };
    }

    if (typeof value === "function") {
        return `[Function: ${value.name || "anonymous"}]`;
    }

    if (typeof value === "bigint") {
        return `${value}n`;
    }

    if (value === undefined) {
        return "[undefined]";
    }

    try {
        return structuredClone(value);
    } catch {
        try {
            return JSON.parse(JSON.stringify(value));
        } catch {
            return String(value);
        }
    }
}

function createEmbeddedConsole({

    showCopyButtons = false,

    logTestOutput = true
} = {}) {

        typeof mountPoint === "string"
            ? document.querySelector(mountPoint)
            : mountPoint;

    if (!(mount instanceof Element)) {
        throw new TypeError(
            "Embedded mode requires mountPoint to be an element or selector."
        );
    }

    const host = document.createElement("div");
    host.className = "mobile-dev-console-host";

    Object.assign(host.style, {
        position: "relative",
        width: "100%",
        height,
        minHeight: "0",

        if (event.origin !== frameUrl.origin) return;
        if (event.data?.source !== CONSOLE_SOURCE) return;

        if (event.data.type === "console-collapse-changed") {

            host.style.height = collapsed ? "40px" : height;

    const queue = [];

    const pendingRegistryRequests = new Map();

        "info",
        "debug",
        "warn",
        "error",
        "group",
        "groupCollapsed",
        "groupEnd",
        "table",
        "clear"

    const installedWrappers = {};
    let devConsole;

    function postToFrame(message) {
        if (destroyed) return;

        const payload = {
            source: CONSOLE_SOURCE,
            ...message

        if (!initialized) {
            queue.push(payload);
            return;

    let inspectorHost = null;

        postToFrame({
            type: "inspect-event",
            event,
            payload
        });

    inspectorHost = new DOMInspectorHost({
        document: window.document,
        mountElement: host,

        onPick({ id, details }) {
            sendInspectorEvent("picked", { id, details });

        onPickCancelled() {
            sendInspectorEvent("pick-cancelled");
        },

        onSelectionChange({ id, details }) {
            sendInspectorEvent("selection-changed", {
                id,
                details
            });
        }

    function handleInspectorRequest(message) {
        const { requestId, action, payload = {} } = message;

        try {
            let result;

            switch (action) {
                case "get-tree":
                    result = inspectorHost.getTree();
                    break;

                case "get-details":
                    result = inspectorHost.getDetails(payload.id);
                    break;

                case "select":
                    result = inspectorHost.select(payload.id);
                    break;

                case "clear-selection":
                    inspectorHost.clearSelection();
                    result = true;
                    break;

                case "start-picking":
                    inspectorHost.startPicking();
                    result = true;
                    break;

                case "cancel-picking":
                    inspectorHost.cancelPicking();
                    result = true;
                    break;

                default:
                    throw new Error(`Unknown inspector action: ${action}`);
            }

            postToFrame({
                type: "inspect-response",
                requestId,
                action,
                ok: true,
                result
            });
        } catch (error) {
            postToFrame({
                type: "inspect-response",
                requestId,
                action,
                ok: false,
                error: error?.message || String(error)
            });
        }
    }

    function flushQueue() {
        while (queue.length && initialized && !destroyed) {
            frame.contentWindow?.postMessage(queue.shift(), targetOrigin);
        }

    function send(method, args = []) {
        postToFrame({
            type: "console-call",
            method,
            args: args.map(serialize)
        });

    function makeRequestId(prefix) {
        nextRequestId += 1;
        return `${prefix}-${nextRequestId}`;

    function validateSuite(definition) {
        if (!definition || typeof definition !== "object") {
            throw new TypeError("registerSuite() expects a suite object.");
        }

        if (typeof definition.id !== "string" || !definition.id.trim()) {
            throw new TypeError("Suite id must be a non-empty string.");
        }

        if (typeof definition.name !== "string" || !definition.name.trim()) {
            throw new TypeError("Suite name must be a non-empty string.");
        }

        if (!Array.isArray(definition.actions)) {
            throw new TypeError("Suite actions must be an array.");
        }

        const actionIds = new Set();

        for (const action of definition.actions) {
            if (!action || typeof action.id !== "string" || !action.id.trim()) {
                throw new TypeError("Every action needs a non-empty id.");
            }

            if (actionIds.has(action.id)) {
                throw new Error(
                    `Duplicate action id "${action.id}" in suite "${definition.id}".`
                );
            }

            actionIds.add(action.id);

            if (typeof action.label !== "string" || !action.label.trim()) {
                throw new TypeError(`Action "${action.id}" needs a label.`);
            }

            if (typeof action.run !== "function") {
                throw new TypeError(
                    `Action "${action.id}" needs a run() function.`
                );
            }
        }
    }

    function getSuiteMetadata(suite) {
        return {
            id: suite.id,
            name: suite.name,
            description: suite.description ?? "",
            actions: suite.actions.map(action => ({
                id: action.id,
                label: action.label,
                description: action.description ?? ""
            }))
        };
    }

    function registerSuite(definition) {
        validateSuite(definition);

        if (suites.has(definition.id)) {
            throw new Error(`Suite "${definition.id}" is already registered.`);
        }

        // Keep callbacks in the host page. Never try to transfer functions
        // through postMessage.
        const suite = {
            id: definition.id,
            name: definition.name,
            description: definition.description ?? "",
            actions: definition.actions.map(action => ({
                id: action.id,
                label: action.label,
                description: action.description ?? "",
                run: action.run
            }))
        };

        suites.set(suite.id, suite);

        postToFrame({
            type: "suite-register",
            suite: getSuiteMetadata(suite)
        });

        // Matches the familiar TestRegistry disposer pattern.
        return () => unregisterSuite(suite.id);
    }

    function unregisterSuite(id) {
        if (!suites.has(id)) {
            return false;
        }

        suites.delete(id);

        postToFrame({
            type: "suite-unregister",
            suiteId: id
        });

        return true;
    }

    function getSuites() {
        return new Promise((resolve, reject) => {
            if (destroyed) {
                reject(new Error("Developer console has been destroyed."));
                return;
            }

            const requestId = makeRequestId("registry");

            pendingRegistryRequests.set(requestId, {
                resolve,
                reject
            });

            postToFrame({
                type: "registry-get-suites",
                requestId
            });
        });
    }

    async function runHostAction(message) {
        const { requestId, suiteId, actionId } = message;

        const suite = suites.get(suiteId);
        const action = suite?.actions.find(item => item.id === actionId);

        if (!suite || !action) {
            postToFrame({
                type: "action-result",
                requestId,
                ok: false,
                error: {
                    name: "Error",
                    message: `Action "${suiteId}/${actionId}" is not registered.`
                }
            });
            return;
        }

        try {
            const result = await action.run({
                console: devConsole,
                suite: {
                    id: suite.id,
                    name: suite.name
                },
                action: {
                    id: action.id,
                    label: action.label
                }
            });

            postToFrame({
                type: "action-result",
                requestId,
                ok: true,
                resultUndefined: result === undefined,
                result: serialize(result)
            });
        } catch (error) {
            postToFrame({
                type: "action-result",
                requestId,
                ok: false,
                error: serialize(error)
            });
        }
    }

    function onMessage(event) {
        if (event.source !== frame.contentWindow) return;
        if (event.origin !== frameUrl.origin) return;
        if (event.data?.source !== CONSOLE_SOURCE) return;

        const data = event.data;

        if (data.type === "console-mode-toggle-request") {
            onModeToggle?.();
            return;
        }

        if (data.type === "console-ready") {
            if (ready) return;

            ready = true;

            // Send initialization directly. It must not wait in the
            // regular command queue, because that queue waits for initialization.
            frame.contentWindow?.postMessage(
                {
                    source: CONSOLE_SOURCE,
                    type: "console-init",
                    settings: {
                        captureConsole,
                        mirrorToNative,
                        showCopyButtons,
                        allowDragOutsideBounds,
                        logTestOutput
                    }
                },
                targetOrigin
            );

            return;
        }

        if (data.type === "console-initialized") {
            if (!ready || initialized) return;

            initialized = true;
            flushQueue();

            postToFrame({
                type: "console-parent-connected"
            });

            return;
        }

        if (data.type === "console-init-error") {
            const message = ["[Console Initialization Failed]", data.message];

            if (typeof originalConsole.error === "function") {
                originalConsole.error(...message);
            }

            for (const pending of pendingRegistryRequests.values()) {
                pending.reject(
                    new Error(data.message || "Console initialization failed.")
                );
            }

            pendingRegistryRequests.clear();
            queue.length = 0;

            return;
        }

        if (data.type === "console-action-request") {
            void runHostAction(data);
            return;
        }

        if (data.type === "registry-suites-result") {
            const pending = pendingRegistryRequests.get(data.requestId);
            if (!pending) return;

            pendingRegistryRequests.delete(data.requestId);

            if (data.ok) {
                pending.resolve(data.suites);
            } else {
                pending.reject(
                    new Error(data.error?.message || "Could not read suites.")
                );
            }

            return;
        }

        if (data.type === "suite-registration-error") {
            send("error", ["[TESTS] Suite registration failed:", data.message]);
        }

        if (data.type === "inspect-request") {
            handleInspectorRequest(data);
        }
    }

    window.addEventListener("message", onMessage);

    function onError(event) {
        send("error", [
            event.error || {
                message: event.message,
                filename: event.filename,
                line: event.lineno,
                column: event.colno
            }
        ]);
    }

    function onUnhandledRejection(event) {
        send("error", ["Unhandled promise rejection:", event.reason]);
    }

    window.addEventListener("error", onError, true);
    window.addEventListener("unhandledrejection", onUnhandledRejection);

    // Preserve native console output while forwarding it to the iframe.
    if (captureConsole) {
        for (const method of methods) {
            originalConsole[method] = console[method];
            installedWrappers[method] = (...args) => {
                send(method, args);
                if (
                    mirrorToNative &&
                    typeof originalConsole[method] === "function"
                ) {
                    originalConsole[method].apply(console, args);
                }
            };
            console[method] = installedWrappers[method];
        }
    }

    devConsole = {
        log: (...args) => send("log", args),
        info: (...args) => send("info", args),
        debug: (...args) => send("debug", args),
        warn: (...args) => send("warn", args),
        error: (...args) => send("error", args),
        group: (...args) => send("group", args),
        groupCollapsed: (...args) => send("groupCollapsed", args),
        groupEnd: () => send("groupEnd"),
        table: data => send("table", [data]),
        clear: () => send("clear"),
        frame,

        tests: {
            registerSuite,
            unregisterSuite,
            getSuites
        }
    };

    // Compatibility alias only. New integrations should use the returned instance.
    const previousGlobalDevConsole = window.devConsole;
    if (previousGlobalDevConsole == null) {
        window.devConsole = devConsole;
    }

    mount.appendChild(host);

    const api = {
        frame,
        host,
        devConsole,
        mode: "embedded",
        registerSuite,
        unregisterSuite,
        getSuites,

        destroy() {
            if (destroyed) return;
            destroyed = true;

            window.removeEventListener("message", onMessage);
            window.removeEventListener("message", onConsoleMessage);
            window.removeEventListener("error", onError, true);
            window.removeEventListener(
                "unhandledrejection",
                onUnhandledRejection
            );

            for (const method of methods) {
                // Do not overwrite another tool that wrapped console after us.
                if (
                    installedWrappers[method] &&
                    console[method] === installedWrappers[method]
                ) {
                    console[method] = originalConsole[method];
                }
            }

            for (const pending of pendingRegistryRequests.values()) {
                pending.reject(
                    new Error("Developer console has been destroyed.")
                );
            }

            pendingRegistryRequests.clear();
            suites.clear();
            queue.length = 0;

            inspectorHost?.destroy();

            host.remove();

            if (window.devConsole === devConsole) {
                if (previousGlobalDevConsole == null) delete window.devConsole;
                else window.devConsole = previousGlobalDevConsole;
            }
        }
    };

    return api;
}

    bindAutoScroll() {
        this.outputScroll = this.element.querySelector(
            ".console-output-scroll"
        );
        const logElement = this.element.querySelector("#js-console");

        if (!this.outputScroll || !logElement) {
            return;
        }

        // Watch only direct children of the log root. Expanding or collapsing
        // an existing group can change descendants, but must not trigger scrolling.
        // A newly appended top-level entry is the only event that auto-scrolls.
        this.logObserver = new MutationObserver(mutations => {
            const hasNewEntry = mutations.some(
                mutation =>
                    mutation.type === "childList" &&
                    mutation.addedNodes.length > 0
            );

            if (!hasNewEntry) {
                return;
            }

            // Scroll only the bounded log-output viewport, never the page.
            this.outputScroll.scrollTo({
                top: this.outputScroll.scrollHeight,
                behavior: "auto"
            });
        });

        this.logObserver.observe(logElement, {
            childList: true,
            subtree: false
        });
    }

    captureNativeConsole() {
        if (this.nativeConsole) {
            return;
        }

        const methods = [
            "log",
            "info",
            "debug",
            "warn",
            "error",
            "group",
            "groupCollapsed",
            "groupEnd",
            "table"
        ];

        this.nativeConsole = {};

        for (const method of methods) {
            this.nativeConsole[method] = console[method].bind(console);
        }

        this.nativeConsole.clear = console.clear.bind(console);

        for (const method of methods) {
            console[method] = (...items) => {
                if (this.settings.mirrorToNative) {
                    this.nativeConsole[method](...items);
                }

                if (method === "info" || method === "debug") {
                    this.output.log(...items);
                } else {
                    this.output[method](...items);
                }
            };
        }

        console.clear = () => {
            this.output.clear();

            if (this.settings.mirrorToNative) {
                this.nativeConsole.clear();
            }
        };
    }

    restoreNativeConsole() {
        if (!this.nativeConsole) {
            return;
        }

        for (const [method, original] of Object.entries(this.nativeConsole)) {
            console[method] = original;
        }

        this.nativeConsole = null;
    }

    bindDragging() {
        this.isDocked = true;
        this.isDragging = false;
        this.dragOffsetX = 0;
        this.dragOffsetY = 0;

        const onPointerDown = event => {
            if (
                this.settings.mode === "embedded" ||
                (event.button !== 0 && event.pointerType === "mouse")
            ) {
                return;
            }

            // Start dragging from the first press, including when docked.
            // This avoids requiring a separate tap to unlock the console.
            if (this.isDocked) this.undock();
            this.isDragging = true;

            const rect = this.element.getBoundingClientRect();
            this.dragOffsetX = event.clientX - rect.left;
            this.dragOffsetY = event.clientY - rect.top;

            try {
                this.dragHandle.setPointerCapture(event.pointerId);
            } catch {
                // Document-level pointer listeners are the fallback.
            }

            event.preventDefault();
        };

        const onPointerMove = event => {
            if (!this.isDragging || this.isDocked) return;

            const viewport = this.getViewportMetrics();
            const rect = this.element.getBoundingClientRect();
            let x = event.clientX - this.dragOffsetX - viewport.offsetLeft;
            let y = event.clientY - this.dragOffsetY - viewport.offsetTop;
            const width = rect.width || viewport.width;
            const height = rect.height || 44;
            const visibleStrip = Math.min(48, width, height);

            if (this.settings.allowDragOutsideBounds) {
                // Allow most of the console to sit outside the viewport, but
                // always leave a usable strip on-screen so it can be recovered.
                x = Math.min(viewport.width - visibleStrip, Math.max(visibleStrip - width, x));
                y = Math.min(viewport.height - visibleStrip, Math.max(visibleStrip - height, y));
            } else {
                x = Math.min(Math.max(0, viewport.width - width), Math.max(0, x));
                y = Math.min(Math.max(0, viewport.height - height), Math.max(0, y));
            }

            this.viewportPosition = { x, y };
            this.applyViewportPosition();
            event.preventDefault();
        };

        const stopDragging = () => {
            this.isDragging = false;
        };

        this.dragHandle.addEventListener("pointerdown", onPointerDown);
        document.addEventListener("pointermove", onPointerMove, { passive: false });
        document.addEventListener("pointerup", stopDragging);
        document.addEventListener("pointercancel", stopDragging);

        this.cleanupCallbacks.push(() => {
            this.dragHandle.removeEventListener("pointerdown", onPointerDown);
            document.removeEventListener("pointermove", onPointerMove);
            document.removeEventListener("pointerup", stopDragging);
            document.removeEventListener("pointercancel", stopDragging);
        });
    }

    getViewportMetrics() {
        const viewport = window.visualViewport;
        return {
            offsetLeft: viewport?.offsetLeft ?? 0,
            offsetTop: viewport?.offsetTop ?? 0,
            width: Math.max(1, viewport?.width ?? window.innerWidth),
            height: Math.max(1, viewport?.height ?? window.innerHeight)
        };
    }

    bindViewportPositioning() {
        this.viewportPosition = { x: 0, y: 0 };
        this.viewportPositionFrame = 0;

        const refresh = () => {
            if (this.viewportPositionFrame) return;
            this.viewportPositionFrame = requestAnimationFrame(() => {
                this.viewportPositionFrame = 0;
                this.refreshViewportPosition();
            });
        };

        const viewport = window.visualViewport;
        viewport?.addEventListener("resize", refresh);
        viewport?.addEventListener("scroll", refresh);
        window.addEventListener("resize", refresh);
        window.addEventListener("orientationchange", refresh);

        this.cleanupCallbacks.push(() => {
            viewport?.removeEventListener("resize", refresh);
            viewport?.removeEventListener("scroll", refresh);
            window.removeEventListener("resize", refresh);
            window.removeEventListener("orientationchange", refresh);
            if (this.viewportPositionFrame) {
                cancelAnimationFrame(this.viewportPositionFrame);
                this.viewportPositionFrame = 0;
            }
        });

        // Wait one frame so the console has its final initial dimensions.
        refresh();
    }

    refreshViewportPosition() {
        if (this.settings.mode !== "overlay" || !this.element.isConnected) return;
        const viewport = this.getViewportMetrics();
        const rect = this.element.getBoundingClientRect();

        if (this.isDocked) {
            this.viewportPosition = {
                x: 0,
                y: Math.max(0, viewport.height - rect.height)
            };
        } else {
            this.constrainViewportPosition(viewport, rect);
        }

        this.applyViewportPosition();
    }

    constrainViewportPosition(viewport = this.getViewportMetrics(), rect = this.element.getBoundingClientRect()) {
        const width = rect.width || viewport.width;
        const height = rect.height || 44;
        const visibleStrip = Math.min(48, width, height);
        let { x, y } = this.viewportPosition ?? { x: 0, y: 0 };

        if (this.settings.allowDragOutsideBounds) {
            x = Math.min(viewport.width - visibleStrip, Math.max(visibleStrip - width, x));
            y = Math.min(viewport.height - visibleStrip, Math.max(visibleStrip - height, y));
        } else {
            x = Math.min(Math.max(0, viewport.width - width), Math.max(0, x));
            y = Math.min(Math.max(0, viewport.height - height), Math.max(0, y));
        }

        this.viewportPosition = { x, y };
    }

    applyViewportPosition() {
        if (this.settings.mode !== "overlay" || !this.element.isConnected) return;
        const viewport = this.getViewportMetrics();
        const rect = this.element.getBoundingClientRect();

        if (this.isDocked) {
            this.viewportPosition.x = 0;
            this.viewportPosition.y = Math.max(0, viewport.height - rect.height);
        } else {
            this.constrainViewportPosition(viewport, rect);
        }

        this.element.style.left = `${viewport.offsetLeft + this.viewportPosition.x}px`;
        this.element.style.top = `${viewport.offsetTop + this.viewportPosition.y}px`;
        this.element.style.right = "auto";
        this.element.style.bottom = "auto";
        this.element.style.width = `${viewport.width}px`;
        this.element.style.maxHeight = `${viewport.height}px`;
    }

    undock() {
        if (!this.isDocked) return;

        const viewport = this.getViewportMetrics();
        const rect = this.element.getBoundingClientRect();
        this.viewportPosition = {
            x: rect.left - viewport.offsetLeft,
            y: rect.top - viewport.offsetTop
        };
        this.isDocked = false;
        this.applyViewportPosition();
        this.updatePositionButton();
    }

    dock() {
        if (this.isDocked) return;
        this.isDragging = false;
        this.isDocked = true;
        this.applyViewportPosition();
        this.updatePositionButton();
    }

    async copyLog() {
        const logElement = this.element.querySelector("#js-console");
        const text = logElement?.innerText?.trimEnd() || "";
        const button = this.copyLogButton;
        const label = button.querySelector("span");

        try {
            if (navigator.clipboard?.writeText) {
                await navigator.clipboard.writeText(text);
            } else {
                const textarea = document.createElement("textarea");
                textarea.value = text;
                textarea.setAttribute("readonly", "");
                textarea.style.position = "fixed";
                textarea.style.opacity = "0";
                document.body.appendChild(textarea);
                textarea.select();
                const copied = document.execCommand("copy");
                textarea.remove();

                if (!copied) {
                    throw new Error("Clipboard copy is not available.");
                }
            }

            label.textContent = "COPIED";
            button.classList.add("copied");
            window.setTimeout(() => {
                label.textContent = "COPY LOG";
                button.classList.remove("copied");
            }, 1400);
        } catch (error) {
            label.textContent = "COPY FAILED";
            window.setTimeout(() => {
                label.textContent = "COPY LOG";
            }, 1800);
        }

                delete window.devConsole;

                window.devConsole = this.previousDevConsole;
            }
        }
    }

export default class ConsoleOutput {
    constructor(container, settings = {}) {
        this.container = container;
        this.groupStack = [];

        this.settings = settings;
    }

    // ==========================================
    // PUBLIC LOG METHODS
    // ==========================================

    log(...items) {
        this.render("log", items);
    }

    warn(...items) {
        this.render("warn", items);
    }

    error(...items) {
        if (items.length === 1 && items[0] instanceof Error) {
            this.renderError(items[0]);
            return;
        }

        this.render("error", items);
    }

    group(...items) {
        this.createGroup(false, items);
    }

    groupCollapsed(...items) {
        this.createGroup(true, items);
    }

    groupEnd() {
        if (this.groupStack.length === 0) {
            return;
        }

        this.groupStack.pop();
    }

    clear() {
        this.container.replaceChildren();
        this.groupStack = [];
    }

    // ==========================================
    // GROUPS
    // ==========================================

    createGroup(collapsed, items) {
        const renderedItems = items.map(item => this.format(item));

        const group = document.createElement("div");
        group.className = "group";

        const source = this.getSourceLocation();
        group.dataset.copySource = source ?? "";

        if (collapsed) {
            group.classList.add("collapsed");
        }

        const header = document.createElement("div");
        header.className = "group-header";

        const arrow = document.createElement("span");
        arrow.className = "group-arrow";
        arrow.textContent = collapsed ? "▶" : "▼";

        const label = document.createElement("span");
        label.className = "group-label";
        label.innerHTML = renderedItems.join(" ");

        header.append(arrow, label);

        header.appendChild(
            this.createCopyButton(() => this.getCopyText(group))
        );

        const content = document.createElement("div");
        content.className = "group-content";

        group.append(header, content);

        header.addEventListener("click", event => {
            if (event.target.closest(".console-copy")) {
                return;
            }

            const isCollapsed = group.classList.toggle("collapsed");
            arrow.textContent = isCollapsed ? "▶" : "▼";
        });

        this.append(group);

        this.groupStack.push({
            group,
            content
        });
    }

    append(element) {
        if (this.groupStack.length > 0) {
            const currentGroup = this.groupStack[this.groupStack.length - 1];

            currentGroup.content.appendChild(element);
            return;
        }

        this.container.appendChild(element);
    }

    // ==========================================
    // TABLE
    // ==========================================

    table(data) {
        if (!data || typeof data !== "object") {
            this.log(data);
            return;
        }

        const entries = Object.entries(data);

        if (entries.length === 0) {
            this.log("(empty table)");
            return;
        }

        // Convert each entry into a row with a consistent shape.
        const rows = entries.map(([index, value]) => {
            if (value !== null && typeof value === "object") {
                const cells = {};

                for (const key of Object.keys(value)) {
                    try {
                        cells[key] = value[key];
                    } catch {
                        cells[key] = "[unreadable]";
                    }
                }

                return { index, cells };
            }

            return {
                index,
                cells: { Value: value }
            };
        });

        // Collect all columns across every row.
        const columns = [];

        for (const row of rows) {
            for (const key of Object.keys(row.cells)) {
                if (!columns.includes(key)) {
                    columns.push(key);
                }
            }
        }

        const table = document.createElement("table");

        // --------------------------------------
        // HEADER
        // --------------------------------------

        const thead = document.createElement("thead");
        const headerRow = document.createElement("tr");

        const indexHeader = document.createElement("th");
        indexHeader.textContent = "(index)";
        headerRow.appendChild(indexHeader);

        for (const column of columns) {
            const th = document.createElement("th");
            th.textContent = column;
            headerRow.appendChild(th);
        }

        thead.appendChild(headerRow);
        table.appendChild(thead);

        // --------------------------------------
        // BODY
        // --------------------------------------

        const tbody = document.createElement("tbody");

        for (const row of rows) {
            const tr = document.createElement("tr");

            const indexCell = document.createElement("td");
            indexCell.textContent = row.index;
            tr.appendChild(indexCell);

            for (const column of columns) {
                const td = document.createElement("td");
                const value = row.cells[column];

                td.textContent =
                    value === undefined && !(column in row.cells)
                        ? ""
                        : this.toPlainText(value);

                tr.appendChild(td);
            }

            tbody.appendChild(tr);
        }

        table.appendChild(tbody);

        // --------------------------------------
        // WRAPPER AND COPY TEXT
        // --------------------------------------

        const wrapper = document.createElement("div");
        wrapper.className = "table";

        wrapper.dataset.copyText = [
            ["(index)", ...columns].join("\t"),
            ...rows.map(row =>
                [
                    row.index,
                    ...columns.map(column =>
                        column in row.cells
                            ? this.toPlainText(row.cells[column])
                            : ""
                    )
                ].join("\t")
            )
        ].join("\n");

        wrapper.dataset.copySource = this.getSourceLocation() ?? "";

        wrapper.appendChild(table);

        wrapper.appendChild(
            this.createCopyButton(() => this.getCopyText(wrapper))
        );

        this.append(wrapper);
    }

    // ==========================================
    // FORMATTING
    // ==========================================

    format(value) {
        if (value === null) {
            return "null";
        }

        if (value === undefined) {
            return "undefined";
        }

        if (value instanceof Error) {
            return this.htmlEncode(
                `${value.name}: ${value.message}\n${value.stack ?? ""}`
            );
        }

        if (typeof value === "string") {
            const visible =
                value.length > 5000 ? value.substring(0, 5000) + "..." : value;

            return this.htmlEncode(visible);
        }

        if (typeof value === "function") {
            return this.htmlEncode(value.toString());
        }

        if (typeof value === "object") {
            try {
                const visibleObject = !Array.isArray(value)
                    ? this.unhideProperties(value)
                    : value;

                const result = JSON.stringify(visibleObject, null, 2);

                return this.htmlEncode(
                    result === undefined ? String(value) : result
                );
            } catch (error) {
                return this.htmlEncode(`[Object: ${error.message}]`);
            }
        }

        return this.htmlEncode(String(value));
    }

    unhideProperties(obj) {
        const result = {};

        for (const key in obj) {
            try {
                result[key] = obj[key];
            } catch {
                result[key] = "[unreadable]";
            }
        }

        return result;
    }

    htmlEncode(value) {
        const element = document.createElement("div");

        element.textContent = String(value);

        return element.innerHTML;
    }

    // ==========================================
    // RENDER
    // ==========================================

    render(type, items, prefix) {
        const renderedItems = items.map(item => this.format(item));
        const plainText = items.map(item => this.toPlainText(item)).join(" ");
        const source = this.getSourceLocation();

        const entry = document.createElement("div");
        entry.className = type;

        // Store clean text and source separately.
        entry.dataset.copyText = prefix ? `${prefix}\n${plainText}` : plainText;

        entry.dataset.copySource = source ?? "";

        if (prefix) {
            const prefixElement = document.createElement("span");

            prefixElement.className = "prefix";
            prefixElement.textContent = prefix;

            entry.appendChild(prefixElement);
            entry.appendChild(document.createTextNode("\n"));
        }

        // Formatted values are HTML-encoded by format().
        const content = document.createElement("span");
        content.innerHTML = renderedItems.join(" ");

        entry.appendChild(content);

        entry.appendChild(this.createCopyButton(() => this.getCopyText(entry)));

        this.append(entry);
    }

    // ==========================================
    // ERROR DETAILS
    // ==========================================

    renderError(error) {
        const source = this.getSourceLocation(error);

        const message = `${error.name}: ${error.message}`;

        const trace = error.stack || message;

        const entry = document.createElement("div");
        entry.className = "error error-entry";

        entry.dataset.copyText = message;
        entry.dataset.copySource = source ?? "";

        // --------------------------------------
        // SUMMARY
        // --------------------------------------

        const header = document.createElement("div");
        header.className = "error-summary";

        const arrow = document.createElement("span");
        arrow.className = "error-arrow";
        arrow.textContent = "▶";

        const label = document.createElement("span");
        label.className = "error-message";
        label.textContent = message;

        const sourceLabel = document.createElement("span");
        sourceLabel.className = "error-source";
        sourceLabel.textContent = source ?? "";

        header.append(arrow, label, sourceLabel);

        header.appendChild(
            this.createCopyButton(() => this.getCopyText(entry))
        );

        // --------------------------------------
        // COLLAPSIBLE TRACE
        // --------------------------------------

        const details = document.createElement("div");
        details.className = "error-trace";
        details.hidden = true;

        const traceText = document.createElement("pre");
        traceText.className = "error-trace-text";
        traceText.textContent = trace;

        details.appendChild(traceText);

        details.appendChild(this.createCopyButton(() => trace));

        // --------------------------------------
        // TOGGLE
        // --------------------------------------

        header.addEventListener("click", event => {
            if (event.target.closest(".console-copy")) {
                return;
            }

            details.hidden = !details.hidden;

            entry.classList.toggle("expanded", !details.hidden);

            arrow.textContent = details.hidden ? "▶" : "▼";
        });

        entry.append(header, details);

        this.append(entry);
    }

    async copyText(text) {
        try {
            await navigator.clipboard.writeText(text);
        } catch (error) {
            this.error("Failed to copy:", error);
        }
    }

    getCopyText(element, depth = 0) {
        const indent = "    ".repeat(depth);

        // Individual logs, warnings, errors, and tables.
        if (element.dataset.copyText !== undefined) {
            const lines = element.dataset.copyText.split("\n");

            const source = element.dataset.copySource;

            if (source && lines.length > 0) {
                lines[0] += ` [${source}]`;
            }

            return lines.map(line => indent + line).join("\n");
        }

        // Groups include their heading and all descendants.
        if (element.classList.contains("group")) {
            const label = element.querySelector(
                ":scope > .group-header .group-label"
            );

            const content = element.querySelector(":scope > .group-content");

            const lines = [];

            if (label?.textContent) {
                const source = element.dataset.copySource;

                const heading = source
                    ? `${label.textContent} [${source}]`
                    : label.textContent;

                lines.push(indent + heading);
            }

            if (content) {
                for (const child of content.children) {
                    const text = this.getCopyText(child, depth + 1);

                    if (text) {
                        lines.push(text);
                    }
                }
            }

            return lines.join("\n");
        }

        // Fallback for other containers.
        return Array.from(element.children)
            .map(child => this.getCopyText(child, depth))
            .filter(Boolean)
            .join("\n");
    }

    toPlainText(value) {
        if (value === null) {
            return "null";
        }

        if (value === undefined) {
            return "undefined";
        }

        if (value instanceof Error) {
            return `${value.name}: ${value.message}\n${value.stack ?? ""}`;
        }

        if (typeof value === "string") {
            return value;
        }

        if (typeof value === "function") {
            return value.toString();
        }

        if (typeof value === "object") {
            try {
                return JSON.stringify(value, null, 2) ?? String(value);
            } catch {
                return "[Unserializable object]";
            }
        }

        return String(value);
    }

    createCopyButton(getText) {
        const button = document.createElement("button");

        button.className = "console-copy";
        button.textContent = "Copy";
        button.type = "button";

        button.addEventListener("click", async event => {
            event.stopPropagation();
            await this.copyText(getText());
        });

        return button;
    }

    getSourceLocation(error = null) {
        const stack = error?.stack || new Error().stack;

        if (!stack) {
            return null;
        }

        const internalFiles = [
            "ConsoleOutput.js",
            "Console.js",
            "errorBootstrap.js"
        ];

        const frames = stack.split("\n").slice(1);

        for (const rawFrame of frames) {
            const frame = rawFrame.trim();

            let functionName = "";
            let url = "";
            let line = "";
            let column = "";

            // Chrome / Chromium / Android WebView:
            // at createLoopTest (http://localhost/file.js:471:9)
            let match = frame.match(/^at\s+(.+?)\s+\((.+):(\d+):(\d+)\)$/);

            if (match) {
                functionName = match[1];
                url = match[2];
                line = match[3];
                column = match[4];
            } else {
                // Chrome anonymous or direct URL frame:
                // at http://localhost/file.js:471:9
                match = frame.match(/^at\s+(.+):(\d+):(\d+)$/);

                if (match) {
                    url = match[1];
                    line = match[2];
                    column = match[3];
                } else {
                    // Firefox:
                    // createLoopTest@http://localhost/file.js:471:9
                    match = frame.match(/^(.*?)@(.+):(\d+):(\d+)$/);

                    if (!match) {
                        continue;
                    }

                    functionName = match[1];
                    url = match[2];
                    line = match[3];
                    column = match[4];
                }
            }

            // Ignore frames belonging to the console infrastructure.
            if (internalFiles.some(file => url.includes(file))) {
                continue;
            }

            // Preserve the original URL and useful source coordinates.
            const location = `${url}:${line}:${column}`;

            return functionName ? `${functionName}@${location}` : location;
        }

        return null;
    }

                    >Loading page…</span>
                </div>

                <div class="dom-inspector-workspace">
                    <section
                        class="dom-tree-pane"
                        aria-label="DOM tree"
                    >
                        <div class="dom-pane-heading">
                            ELEMENT TREE
                        </div>

                        <div
                            class="dom-tree-scroll"
                            id="domTree"
                        ></div>
                    </section>

                    <section
                        class="dom-details-pane"
                        aria-label="Selected element details"
                    >
                        <div class="dom-pane-heading">
                            ELEMENT DETAILS
                        </div>

                        <div
                            class="dom-details-scroll"
                            id="domDetails"
                        >
                            <p class="dom-inspector-empty">
                                Choose an element from the tree
                                or use PICK ELEMENT.
                            </p>
                        </div>
                    </section>
                </div>
            </div>
        `;

        this.pickButton = this.panel.querySelector("#domPickElement");

        this.refreshButton = this.panel.querySelector("#domRefreshTree");

        this.status = this.panel.querySelector("#domInspectorStatus");

        this.tree = this.panel.querySelector("#domTree");

        this.details = this.panel.querySelector("#domDetails");
    }

    bindEvents() {
        this.pickButton.addEventListener("click", () => {
            this.setPicking(!this.picking);
        });

        document.addEventListener(
            "pointerdown",
            (this.onPickingPointerDown = event => {
                if (!this.picking) return;

                // The existing toggle button remains interactive.
                if (
                    event.target === this.pickButton ||
                    this.pickButton.contains(event.target)
                ) {
                    return;
                }

                // Any other interaction inside the console cancels picking.
                this.setPicking(false);
            }),
            true
        );

        this.refreshButton.addEventListener("click", () => {
            this.refresh();
        });

        this.tree.addEventListener("click", event => {
            const toggle = event.target.closest("[data-dom-toggle]");

            if (toggle && this.tree.contains(toggle)) {
                this.toggleBranch(toggle);
                return;
            }

            const select = event.target.closest("[data-dom-select]");

            if (select && this.tree.contains(select)) {
                this.selectElement(select.dataset.domSelect, { reveal: false });
            }
        });
    }

    // --------------------------------------------------
    // Parent bridge
    // --------------------------------------------------

    async request(action, payload = {}) {
        if (this.destroyed) {
            throw new Error("DOMInspector has been destroyed.");
        }
    
        if (!this.parentDOMInspector) {
            throw new Error(
                "DOM inspection is unavailable in this mode."
            );
        }
    
        return this.parentDOMInspector.request(action, payload);
    }

    async refresh({ selectedId = this.selectedId, reveal = false } = {}) {
        const version = ++this.refreshVersion;

        this.status.textContent = "Refreshing page tree…";
        this.refreshButton.disabled = true;

        try {
            const tree = await this.request("get-tree");

            if (this.destroyed || version !== this.refreshVersion) {
                return;
            }

            if (!tree) {
                throw new Error("The parent page returned an empty DOM tree.");
            }

            this.treeData = tree;
            this.renderTree();

            if (selectedId) {
                const found = this.findNode(this.treeData, selectedId);

                if (found) {
                    await this.selectElement(selectedId, {
                        reveal
                    });
                } else {
                    this.selectedId = null;
                    this.selectedDetails = null;
                    this.renderSelectedState();
                    this.renderEmptyDetails();
                    this.status.textContent = "Select an element";
                }
            } else {
                this.status.textContent = "Select an element";
            }

            if (this.countNodes(this.treeData) >= this.maxNodes) {
                const note = document.createElement("div");

                note.className = "dom-inspector-limit";
                note.textContent = `Tree may be limited to ${this.maxNodes} elements.`;

                this.tree.appendChild(note);
            }
        } catch (error) {
            if (!this.destroyed && version === this.refreshVersion) {
                this.status.textContent = "Unable to inspect page";
                this.showError(error);
            }
        } finally {
            if (!this.destroyed && version === this.refreshVersion) {
                this.refreshButton.disabled = false;
            }
        }
    }

    async selectElement(id, { reveal = false } = {}) {
        if (!id || this.destroyed) {
            return;
        }

        this.status.textContent = "Selecting element…";

        try {
            const details = await this.request("select", { id });

            if (this.destroyed) {
                return;
            }

            if (!details) {
                this.status.textContent = "Element no longer exists";

                return;
            }

            this.selectedId = id;
            this.selectedDetails = details;

            this.renderSelectedState();

            if (reveal) {
                this.revealNode(id);
            }

            this.status.textContent = this.describeDetails(details);

            this.renderDetails(details);
        } catch (error) {
            if (!this.destroyed) {
                this.status.textContent = "Unable to select element";

                this.showError(error);
            }
        }
    }

    async setPicking(enabled) {
        if (this.destroyed) {
            return;
        }

        enabled = Boolean(enabled);

        if (this.picking === enabled) {
            return;
        }

        // Update the UI immediately.
        this.picking = enabled;
        this.updatePickingUI();

        try {
            const result = await this.request(
                enabled ? "start-picking" : "cancel-picking"
            );

            console.log(
                `[DOMInspector] ${enabled ? "Start" : "Cancel"} picking result:`,
                result
            );

            // Don't reset picking here. The parent controls its
            // completion through picked/pick-cancelled events.
        } catch (error) {
            console.error("[DOMInspector] Picking request failed:", error);

            this.picking = false;
            this.updatePickingUI();
            this.showError(error);
        }
    }

    // --------------------------------------------------
    // Parent-originated events
    // --------------------------------------------------

    handleInspectorEvent(message) {
        if (this.destroyed || !message) {
            return;
        }

        console.log("[DOMInspector] Parent event:", message);

        const { event, payload = {} } = message;

        switch (event) {
            case "picked":
                this.handlePicked(payload);
                break;

            case "selection-changed":
                this.handleSelectionChanged(payload);
                break;

            case "pick-cancelled":
                this.picking = false;
                this.updatePickingUI();
                break;
        }
    }

    async handlePicked({ id, details }) {
        if (!id) return;

        this.selectedId = id;
        this.selectedDetails = details || null;

        await this.refresh({
            selectedId: id,
            reveal: true
        });

        // Refreshing and selecting must not end picking mode.
        if (this.picking) {
            this.updatePickingUI();
        }
    }

    handleSelectionChanged({ id, details }) {
        this.selectedId = id || null;
        this.selectedDetails = details || null;

        this.renderSelectedState();

        if (details) {
            this.status.textContent = this.picking
                ? "Tap another project element · Cancel to stop"
                : this.describeDetails(details);
        } else {
            this.status.textContent = "Select an element";

            this.renderEmptyDetails();
        }
    }

    updatePickingUI() {
        this.pickButton.classList.toggle("active", this.picking);

        this.pickButton.textContent = this.picking
            ? "CANCEL PICKING"
            : "PICK ELEMENT";

        // Keep the picking toggle interactive, but disable other
        // inspector controls while picking is active.
        this.refreshButton.style.pointerEvents = this.picking ? "none" : "";

        const workspace = this.panel.querySelector(".dom-inspector-workspace");

        if (workspace) {
            workspace.style.pointerEvents = this.picking ? "none" : "";
        }

        this.status.textContent = this.picking
            ? "Tap project elements to inspect them · Cancel to stop"
            : this.selectedDetails
              ? this.describeDetails(this.selectedDetails)
              : "Select an element";
    }

    // --------------------------------------------------
    // Tree rendering
    // --------------------------------------------------

    renderTree() {
        this.tree.replaceChildren();

        if (!this.treeData) {
            return;
        }

        this.tree.appendChild(this.createTreeNode(this.treeData, true));
    }

    createTreeNode(data, expanded = false) {
        const node = document.createElement("div");

        node.className = "dom-tree-node";
        node.dataset.domElementId = data.id;

        const row = document.createElement("div");
        row.className = "dom-tree-row";

        const toggle = document.createElement("button");

        toggle.type = "button";
        toggle.className = "dom-tree-toggle";
        toggle.dataset.domToggle = "true";
        toggle.setAttribute("aria-label", "Expand or collapse children");

        const hasChildren =
            Array.isArray(data.children) && data.children.length > 0;

        toggle.disabled = !hasChildren;
        toggle.classList.toggle("no-children", !hasChildren);

        toggle.setAttribute("aria-expanded", String(hasChildren && expanded));

        toggle.textContent = !hasChildren ? "·" : expanded ? "▾" : "▸";

        row.appendChild(toggle);

        const select = document.createElement("button");

        select.type = "button";
        select.className = "dom-tree-select";
        select.dataset.domSelect = data.id;
        select.textContent = data.label;
        select.title = data.label;

        row.appendChild(select);
        node.appendChild(row);

        if (hasChildren) {
            const branch = document.createElement("div");

            branch.className = "dom-tree-children";
            branch.hidden = !expanded;

            for (const child of data.children) {
                branch.appendChild(this.createTreeNode(child, false));
            }

            node.appendChild(branch);
        }

        return node;
    }

    toggleBranch(toggle) {
        const node = toggle.closest(".dom-tree-node");

        const branch = node?.querySelector(":scope > .dom-tree-children");

        if (!branch) {
            return;
        }

        const expanded = toggle.getAttribute("aria-expanded") === "true";

        toggle.setAttribute("aria-expanded", String(!expanded));

        branch.hidden = expanded;
        toggle.textContent = expanded ? "▸" : "▾";
    }

    findNode(node, id) {
        if (!node) {
            return null;
        }

        if (node.id === id) {
            return node;
        }

        for (const child of node.children || []) {
            const found = this.findNode(child, id);

            if (found) {
                return found;
            }
        }

        return null;
    }

    countNodes(node) {
        if (!node) {
            return 0;
        }

        return (
            1 +
            (node.children || []).reduce(
                (count, child) => count + this.countNodes(child),
                0
            )
        );
    }

    revealNode(id) {
        const button = Array.from(
            this.tree.querySelectorAll("[data-dom-select]")
        ).find(item => item.dataset.domSelect === id);

        if (!button) {
            return;
        }

        let branch = button.closest(".dom-tree-children");

        while (branch && branch !== this.tree) {
            branch.hidden = false;

            const parentNode = branch.parentElement?.closest(".dom-tree-node");

            const toggle = parentNode?.querySelector(
                ":scope > .dom-tree-row [data-dom-toggle]"
            );

            if (toggle) {
                toggle.textContent = "▾";
                toggle.setAttribute("aria-expanded", "true");
            }

            branch =
                parentNode?.parentElement?.closest(".dom-tree-children") ||
                null;
        }

        const scroller = this.tree;

        const itemRect = button.getBoundingClientRect();
        const scrollRect = scroller.getBoundingClientRect();

        if (itemRect.top < scrollRect.top) {
            scroller.scrollTop -= scrollRect.top - itemRect.top;
        } else if (itemRect.bottom > scrollRect.bottom) {
            scroller.scrollTop += itemRect.bottom - scrollRect.bottom;
        }
    }

    renderSelectedState() {
        this.tree
            .querySelectorAll(".dom-tree-select.selected")
            .forEach(button => {
                button.classList.remove("selected");
            });

        if (!this.selectedId) {
            return;
        }

        const button = Array.from(
            this.tree.querySelectorAll("[data-dom-select]")
        ).find(item => item.dataset.domSelect === this.selectedId);

        button?.classList.add("selected");
    }

    // --------------------------------------------------
    // Details rendering
    // --------------------------------------------------

    renderDetails(details) {
        this.details.replaceChildren();

        const element = details.element || {};
        const layout = details.layout || {};
        const style = details.computedStyle || {};

        this.addDetailSection("ELEMENT", [
            ["Tag", element.tag || "—"],
            ["ID", element.id || "—"],
            ["Classes", element.className || "—"],
            ["Text", element.text || "—"]
        ]);

        this.addDetailSection("LAYOUT", [
            ["Position", `${layout.x ?? "—"}, ${layout.y ?? "—"}`],
            ["Size", `${layout.width ?? "—"} × ${layout.height ?? "—"}`],
            ["CSS position", layout.position || "—"]
        ]);

        this.addDetailSection("COMPUTED STYLE", [
            ["display", style.display || "—"],
            ["position", style.position || "—"],
            ["color", style.color || "—"],
            ["background", style.backgroundColor || "—"],
            ["font-size", style.fontSize || "—"],
            ["font-family", style.fontFamily || "—"],
            ["font-weight", style.fontWeight || "—"],
            ["margin", style.margin || "—"],
            ["padding", style.padding || "—"],
            ["overflow", style.overflow || "—"],
            ["opacity", style.opacity || "—"]
        ]);

        if (details.attributes?.length) {
            this.addDetailSection(
                "ATTRIBUTES",
                details.attributes.map(attribute => [
                    attribute.name,
                    attribute.value
                ])
            );
        }
    }

    addDetailSection(title, rows) {
        const section = document.createElement("section");
        section.className = "dom-detail-section";

        const heading = document.createElement("h4");
        heading.textContent = title;

        section.appendChild(heading);

        const list = document.createElement("dl");

        for (const [name, value] of rows) {
            const term = document.createElement("dt");
            term.textContent = name;

            const description = document.createElement("dd");
            description.textContent = String(value ?? "—");

            list.append(term, description);
        }

        section.appendChild(list);
        this.details.appendChild(section);
    }

    renderEmptyDetails() {
        this.details.replaceChildren();

        const message = document.createElement("p");
        message.className = "dom-inspector-empty";
        message.textContent =
            "Choose an element from the tree or use PICK ELEMENT.";

        this.details.appendChild(message);
    }

    describeDetails(details) {
        const element = details?.element;

        if (!element) {
            return "Select an element";
        }

        let label = element.tag || "element";

        if (element.id) {
            label += `#${element.id}`;
        }

        if (element.className) {
            label +=
                "." +
                element.className.trim().split(/\s+/).slice(0, 3).join(".");
        }

        return label;
    }

    showError(error) {
        this.details.replaceChildren();

        const message = document.createElement("p");
        message.className = "dom-inspector-empty";
        message.textContent = error?.message || String(error);

        this.details.appendChild(message);
    }

    // --------------------------------------------------
    // Cleanup
    // --------------------------------------------------

    destroy() {
        if (this.destroyed) {
            return;
        }

        this.destroyed = true;
        this.refreshVersion++;

        window.removeEventListener(
            "devtools-inspect-event",
            this.onInspectorEvent
        );

        if (this.picking) {
            this.parentDOMInspector
                ?.request("cancel-picking")
                .catch(() => {});
        }

        this.picking = false;
        this.treeData = null;
        this.selectedDetails = null;
        this.selectedId = null;

        this.panel.replaceChildren();

        document.removeEventListener(
            "pointerdown",
            this.onPickingPointerDown,
            true
        );
    }
}

/**
 * Parent-page DOM inspection engine.
 *
 * Runs in the host page, not inside the console iframe.
 * All returned tree/details data is serializable and safe to
 * send across postMessage.
 */
export default class DOMInspectorHost {
    constructor({
        document: targetDocument = document,
        mountElement = null,
        maxNodes = 500,
        maxDepth = 40,
        onPick = null,
        onPickCancelled = null,
        onSelectionChange = null
    } = {}) {
        if (!targetDocument?.body) {
            throw new Error(
                "DOMInspectorHost requires a document with a body."
            );
        }

        this.document = targetDocument;
        this.window = targetDocument.defaultView;
        this.mountElement = mountElement;
        this.maxNodes = maxNodes;
        this.maxDepth = maxDepth;

        this.onPick = onPick;
        this.onPickCancelled = onPickCancelled;
        this.onSelectionChange = onSelectionChange;

        this.elementIds = new WeakMap();
        this.elementsById = new Map();
        this.nextElementId = 1;

        this.nodeCount = 0;
        this.selectedId = null;
        this.picking = false;
        this.destroyed = false;

        this.highlight = this.createHighlight();

        this.handlePointerDown = this.handlePointerDown.bind(this);
        this.handleKeyDown = this.handleKeyDown.bind(this);
        this.updateHighlight = this.updateHighlight.bind(this);

        this.document.addEventListener(
            "pointerdown",
            this.handlePointerDown,
            true
        );

        this.document.addEventListener("keydown", this.handleKeyDown, true);

        this.window?.addEventListener("resize", this.updateHighlight);

        this.document.addEventListener("scroll", this.updateHighlight, true);
    }

    // --------------------------------------------------
    // Request adapter
    // --------------------------------------------------

    // DOMInspector uses this same request contract in both modes. In embedded
    // mode requests cross postMessage; in overlay mode they call this host.
    async request(action, payload = {}) {
        if (this.destroyed) {
            throw new Error("DOMInspectorHost has been destroyed.");
        }

        switch (action) {
            case "get-tree":
                return this.getTree();
            case "get-details":
                return this.getDetails(payload.id);
            case "select":
                return this.select(payload.id);
            case "clear-selection":
                this.clearSelection();
                return true;
            case "start-picking":
                this.startPicking();
                return true;
            case "cancel-picking":
                this.cancelPicking();
                return true;
            default:
                throw new Error(`Unknown inspector action: ${action}`);
        }
    }

    // --------------------------------------------------
    // DOM tree
    // --------------------------------------------------

    getTree() {
        this.nodeCount = 0;

        const body = this.document.body;

        if (!body) {
            return null;
        }

        // Always retain body as the root, even if it contains
        // the DevTools mount point.
        this.nodeCount++;

        const children = [];

        for (const child of body.children) {
            if (this.nodeCount >= this.maxNodes) {
                break;
            }

            const node = this.buildNode(child, 1);

            if (node) {
                children.push(node);
            }
        }

        return {
            id: this.getElementId(body),
            label: this.describeElement(body),
            tag: "body",
            elementId: body.id || "",
            className: this.getClassName(body),
            hasChildren: children.length > 0,
            children
        };
    }

    buildNode(element, depth) {
        if (
            !element ||
            element.nodeType !== 1 ||
            this.isExcluded(element) ||
            this.nodeCount >= this.maxNodes
        ) {
            return null;
        }

        this.nodeCount++;

        const id = this.getElementId(element);
        const children = [];

        if (depth < this.maxDepth) {
            for (const child of element.children) {
                if (this.nodeCount >= this.maxNodes) {
                    break;
                }

                const node = this.buildNode(child, depth + 1);

                if (node) {
                    children.push(node);
                }
            }
        }

        return {
            id,
            label: this.describeElement(element),
            tag: element.tagName.toLowerCase(),
            elementId: element.id || "",
            className: this.getClassName(element),
            hasChildren: element.children.length > 0,
            children
        };
    }

    getElementId(element) {
        let id = this.elementIds.get(element);

        if (!id) {
            id = `dom-${this.nextElementId++}`;

            this.elementIds.set(element, id);
            this.elementsById.set(id, element);
        }

        return id;
    }

    getElementById(id) {
        const element = this.elementsById.get(id);

        if (!element || !element.isConnected || this.isExcluded(element)) {
            return null;
        }

        return element;
    }

    describeElement(element) {
        let label = element.tagName.toLowerCase();

        if (element.id) {
            label += `#${element.id}`;
        }

        const className = this.getClassName(element);

        if (className) {
            label += "." + className.trim().split(/\s+/).slice(0, 3).join(".");
        }

        return label;
    }

    getClassName(element) {
        if (typeof element.className === "string") {
            return element.className;
        }

        return element.getAttribute("class") || "";
    }

    isExcluded(element) {
        if (!element || element.nodeType !== 1) {
            return true;
        }

        if (element.closest?.("[data-devtools-pick-cancel]")) {
            return true;
        }

        // Exclude the supplied mount element and its descendants.
        if (
            this.mountElement &&
            (element === this.mountElement ||
                this.mountElement.contains(element))
        ) {
            return true;
        }

        // Exclude the console's outer mount container, too.
        if (element.id === "devtools") {
            return true;
        }

        if (element.hasAttribute("data-dom-inspector-highlight")) {
            return true;
        }

        return [
            "SCRIPT",
            "STYLE",
            "LINK",
            "META",
            "TITLE",
            "NOSCRIPT"
        ].includes(element.tagName);
    }

    // --------------------------------------------------
    // Element details
    // --------------------------------------------------

    getDetails(id) {
        const element = this.getElementById(id);

        if (!element) {
            return null;
        }

        const style = this.window.getComputedStyle(element);
        const rect = element.getBoundingClientRect();

        const attributes = Array.from(element.attributes).map(attribute => ({
            name: attribute.name,
            value: attribute.value
        }));

        return {
            id,
            element: {
                tag: element.tagName.toLowerCase(),
                id: element.id || "",
                className: this.getClassName(element),
                text: this.getText(element)
            },

            layout: {
                position: style.position,
                x: this.round(rect.x),
                y: this.round(rect.y),
                width: this.round(rect.width),
                height: this.round(rect.height)
            },

            computedStyle: {
                display: style.display,
                position: style.position,
                color: style.color,
                backgroundColor: style.backgroundColor,
                fontSize: style.fontSize,
                fontFamily: style.fontFamily,
                fontWeight: style.fontWeight,
                margin: style.margin,
                padding: style.padding,
                overflow: style.overflow,
                opacity: style.opacity
            },

            attributes
        };
    }

    getText(element) {
        return (element.innerText || element.textContent || "")
            .replace(/\s+/g, " ")
            .trim()
            .slice(0, 500);
    }

    round(value) {
        return Math.round(value * 100) / 100;
    }

    // --------------------------------------------------
    // Selection and highlighting
    // --------------------------------------------------

    select(id) {
        const element = this.getElementById(id);

        if (!element) {
            this.clearSelection();
            return null;
        }

        this.selectedId = id;
        this.updateHighlight();

        const details = this.getDetails(id);

        this.onSelectionChange?.({
            id,
            details
        });

        return details;
    }

    clearSelection() {
        this.selectedId = null;

        if (this.highlight) {
            this.highlight.style.display = "none";
        }

        this.onSelectionChange?.({
            id: null,
            details: null
        });
    }

    createHighlight() {
        const highlight = this.document.createElement("div");

        highlight.setAttribute("data-dom-inspector-highlight", "");

        Object.assign(highlight.style, {
            position: "fixed",
            display: "none",
            pointerEvents: "none",
            zIndex: "2147483646",
            boxSizing: "border-box",
            border: "2px solid #38bdf8",
            background: "rgba(56, 189, 248, 0.12)",
            borderRadius: "2px",
            margin: "0",
            padding: "0"
        });

        this.document.body.appendChild(highlight);

        return highlight;
    }

    updateHighlight() {
        if (!this.highlight || !this.selectedId) {
            return;
        }

        const element = this.getElementById(this.selectedId);

        if (!element) {
            this.highlight.style.display = "none";
            return;
        }

        const rect = element.getBoundingClientRect();

        Object.assign(this.highlight.style, {
            display: "block",
            left: `${rect.left}px`,
            top: `${rect.top}px`,
            width: `${rect.width}px`,
            height: `${rect.height}px`
        });
    }

    // --------------------------------------------------
    // Element picking
    // --------------------------------------------------

    startPicking() {
        if (this.destroyed || this.picking) {
            return;
        }

        this.picking = true;

        this.document.documentElement.classList.add("dom-inspector-picking");
    }

    cancelPicking() {
        if (!this.picking) {
            return;
        }

        this.picking = false;

        this.document.documentElement.classList.remove("dom-inspector-picking");

        this.onPickCancelled?.();
    }

    handlePointerDown(event) {
        if (!this.picking || this.destroyed) {
            return;
        }

        const target = event.target;

        if (
            !target ||
            target.nodeType !== 1 ||
            target === this.document.documentElement ||
            target === this.document.body
        ) {
            this.cancelPicking();
            return;
        }

        // Overlay controls must remain usable while picking so the user can
        // press CANCEL PICKING. Let the inspector UI handle console clicks.
        if (
            this.mountElement &&
            (target === this.mountElement || this.mountElement.contains(target))
        ) {
            return;
        }

        // Taps outside the inspectable project area cancel picking.
        if (this.isExcluded(target)) {
            this.cancelPicking();
            return;
        }

        // Select the element without activating the underlying page.
        event.preventDefault();
        event.stopPropagation();
        event.stopImmediatePropagation();

        const id = this.getElementId(target);
        const details = this.select(id);

        // IMPORTANT: do not clear this.picking here.
        // The user can select another element immediately.
        this.onPick?.({
            id,
            details
        });
    }

    handleKeyDown(event) {
        if (!this.picking || event.key !== "Escape") {
            return;
        }

        event.preventDefault();
        event.stopPropagation();
        event.stopImmediatePropagation();

        this.cancelPicking();
    }

    // --------------------------------------------------
    // Cleanup
    // --------------------------------------------------

    destroy() {
        if (this.destroyed) {
            return;
        }

        this.destroyed = true;
        this.picking = false;

        this.document.documentElement.classList.remove("dom-inspector-picking");

        this.document.removeEventListener(
            "pointerdown",
            this.handlePointerDown,
            true
        );

        this.document.removeEventListener("keydown", this.handleKeyDown, true);

        this.window?.removeEventListener("resize", this.updateHighlight);

        this.document.removeEventListener("scroll", this.updateHighlight, true);

        this.highlight?.remove();

        this.highlight = null;
        this.selectedId = null;

        this.elementsById.clear();
        this.elementIds = new WeakMap();
    }
}

(() => {
    // Prevent accidental duplicate installation.
    if (window.__errorBootstrapInstalled) {
        return;
    }

    window.__errorBootstrapInstalled = true;

    // --------------------------------------------------
    // DISPLAY FATAL ERROR
    // --------------------------------------------------

    window.showFatalError = function (title, error) {
        const panel = document.getElementById("productionErrorLog");

        if (!panel) {
            return;
        }

        let message;

        if (error instanceof Error) {
            message = error.stack || error.message;
        } else if (error && typeof error === "object") {
            try {
                message = JSON.stringify(error, null, 2);
            } catch {
                message = String(error);
            }
        } else {
            message = String(error ?? "Unknown error");
        }

        // Build the panel using DOM methods only.
        // This does not depend on console.js or console.css.

        panel.replaceChildren();

        Object.assign(panel.style, {
            display: "block",
            position: "fixed",
            top: "10px",
            left: "10px",
            right: "10px",
            maxHeight: "70vh",
            overflow: "auto",
            padding: "12px",
            background: "#300",
            color: "#fff",
            border: "2px solid #f55",
            borderRadius: "6px",
            fontFamily: "monospace",
            fontSize: "13px",
            whiteSpace: "pre-wrap",
            overflowWrap: "anywhere",
            zIndex: "2147483647",
            boxSizing: "border-box"
        });

        const heading = document.createElement("strong");

        heading.textContent = `⚠ ${title}`;

        const details = document.createElement("pre");

        Object.assign(details.style, {
            margin: "10px 0 0",
            color: "#ffaaaa",
            font: "inherit",
            whiteSpace: "pre-wrap",
            overflowWrap: "anywhere"
        });

        details.textContent = message;

        panel.append(heading, details);
    };

    // --------------------------------------------------
    // RUNTIME AND RESOURCE ERRORS
    // --------------------------------------------------

    window.addEventListener(
        "error",
        event => {
            const title = event.error
                ? "JavaScript Runtime Error"
                : "JavaScript / Module / Resource Load Error";

            const error = event.error || {
                message:
                    event.message ||
                    "A script, module, or other resource failed to load.",
                filename: event.filename || null,
                line: event.lineno || null,
                column: event.colno || null,
                target:
                    event.target?.src ||
                    event.target?.href ||
                    event.target?.tagName ||
                    null
            };

            if (window.devConsole) {
                // The main console is running: use it exclusively.
                window.devConsole.error(error);
                return;
            }

            // The main console is unavailable: use the fallback.
            window.showFatalError(title, error);

// TEMP
            // Also report the error to the parent loader.
            window.parent.postMessage(
                {
                    source: "mobile-dev-console",
                    type: "console-bootstrap-error",
                    title,
                    error:
                        error instanceof Error
                            ? {
                                  name: error.name,
                                  message: error.message,
                                  stack: error.stack
                              }
                            : error
                },
                location.origin
            );
        },
        true
    );

    // --------------------------------------------------
    // UNHANDLED PROMISE ERRORS
    // --------------------------------------------------

    window.addEventListener("unhandledrejection", event => {
        if (window.devConsole) {
            window.devConsole.error(event.reason);
            return;
        }

        window.showFatalError("Unhandled Promise Error", event.reason);

// TEMP
        window.parent.postMessage(
            {
                source: "mobile-dev-console",
                type: "console-bootstrap-error",
                title: "Unhandled Promise Error",
                error:
                    event.reason instanceof Error
                        ? {
                              name: event.reason.name,
                              message: event.reason.message,
                              stack: event.reason.stack
                          }
                        : String(event.reason)
            },
            location.origin
        );
    });

/**
 * Public registry for project-provided debug/test actions.
 *
 * Projects register suites through window.devConsole.tests.registerSuite().
 * The console owns presentation; suites own the behavior of their actions.
 */
export default class TestRegistry {
    constructor(devConsole, panel) {
        this.devConsole = devConsole;
        this.panel = panel;
        this.suites = new Map();

        this.render();
    }

    registerSuite(definition) {
        this.validateSuite(definition);

        if (this.suites.has(definition.id)) {
            throw new Error(
                `TestRegistry: suite "${definition.id}" is already registered.`
            );
        }

        // Keep a normalized copy so external mutation cannot silently change the UI.
        const suite = {
            id: definition.id,
            name: definition.name,
            description: definition.description ?? "",
            actions: definition.actions.map(action => ({
                id: action.id,
                label: action.label,
                description: action.description ?? "",
                run: action.run
            }))
        };

        this.suites.set(suite.id, suite);
        this.render();

        // Return a convenient disposer for module/app lifecycle cleanup.
        return () => this.unregisterSuite(suite.id);
    }

    unregisterSuite(id) {
        const removed = this.suites.delete(id);

        if (removed) {
            this.render();
        }

        return removed;
    }

    getSuites() {
        return Array.from(this.suites.values(), suite => ({
            id: suite.id,
            name: suite.name,
            description: suite.description,
            actions: suite.actions.map(({ id, label, description }) => ({
                id,
                label,
                description
            }))
        }));
    }

    validateSuite(suite) {
        if (!suite || typeof suite !== "object") {
            throw new TypeError(
                "TestRegistry: registerSuite() expects a suite object."
            );
        }

        if (typeof suite.id !== "string" || !suite.id.trim()) {
            throw new TypeError(
                "TestRegistry: suite.id must be a non-empty string."
            );
        }

        if (typeof suite.name !== "string" || !suite.name.trim()) {
            throw new TypeError(
                "TestRegistry: suite.name must be a non-empty string."
            );
        }

        if (!Array.isArray(suite.actions)) {
            throw new TypeError(
                "TestRegistry: suite.actions must be an array."
            );
        }

        const actionIds = new Set();

        for (const action of suite.actions) {
            if (!action || typeof action !== "object") {
                throw new TypeError(
                    `TestRegistry: invalid action in suite "${suite.id}".`
                );
            }

            if (typeof action.id !== "string" || !action.id.trim()) {
                throw new TypeError(
                    `TestRegistry: every action in "${suite.id}" needs an id.`
                );
            }

            if (actionIds.has(action.id)) {
                throw new Error(
                    `TestRegistry: duplicate action id "${action.id}" in suite "${suite.id}".`
                );
            }
            actionIds.add(action.id);

            if (typeof action.label !== "string" || !action.label.trim()) {
                throw new TypeError(
                    `TestRegistry: action "${action.id}" needs a label.`
                );
            }

            if (typeof action.run !== "function") {
                throw new TypeError(
                    `TestRegistry: action "${action.id}" needs a run() function.`
                );
            }
        }
    }

    async runAction(suite, action, button, status) {
        if (button.disabled) {
            return;
        }

        button.disabled = true;
        button.classList.add("running");
        const originalLabel = button.textContent;
        button.textContent = "RUNNING…";
        status.textContent = `${suite.name} / ${action.label}`;

        try {
            const result = await action.run({
                console: this.devConsole,
                suite: { id: suite.id, name: suite.name },
                action: { id: action.id, label: action.label }
            });

            if (this.devConsole.settings.logTestOutput) {
                if (result !== undefined) {
                    this.devConsole.log(
                        `[TESTS] ${suite.name} → ${action.label}`,
                        result
                    );
                } else {
                    this.devConsole.log(
                        `[TESTS] Completed: ${suite.name} → ${action.label}`
                    );
                }
            }

            status.textContent = `Completed: ${action.label}`;
            button.textContent = "DONE";
        } catch (error) {
            this.devConsole.error(error);
            status.textContent = `Failed: ${action.label}`;
            button.textContent = "FAILED";
        } finally {
            button.classList.remove("running");
            window.setTimeout(() => {
                if (!button.isConnected) return;
                button.textContent = originalLabel;
                button.disabled = false;
            }, 1200);
        }
    }

    render() {
        if (!this.panel) return;

        const root = document.createElement("div");
        root.className = "console-tests";

        const toolbar = document.createElement("div");
        toolbar.className = "console-tests-heading";

        const title = document.createElement("strong");
        title.textContent = "REGISTERED TOOLS";

        const count = document.createElement("span");
        count.className = "console-tests-count";
        count.textContent = `${this.suites.size} SUITE${this.suites.size === 1 ? "" : "S"}`;
        toolbar.append(title, count);
        root.appendChild(toolbar);

        const status = document.createElement("div");
        status.className = "console-tests-status";
        status.setAttribute("role", "status");
        status.textContent = this.suites.size
            ? "Choose an action to run."
            : "No project tools registered yet.";
        root.appendChild(status);

        if (this.suites.size === 0) {
            const empty = document.createElement("div");
            empty.className = "console-tests-empty";
            empty.textContent =
                "Register a suite from an integration module to make its tools available here.";
            root.appendChild(empty);
        }

        for (const suite of this.suites.values()) {
            const section = document.createElement("section");
            section.className = "console-tests-suite";
            section.dataset.suiteId = suite.id;

            const heading = document.createElement("div");
            heading.className = "console-tests-suite-heading";

            const name = document.createElement("strong");
            name.textContent = suite.name;
            heading.appendChild(name);

            if (suite.description) {
                const description = document.createElement("p");
                description.textContent = suite.description;
                heading.appendChild(description);
            }

            section.appendChild(heading);

            for (const action of suite.actions) {
                const row = document.createElement("div");
                row.className = "console-tests-action";

                const info = document.createElement("div");
                info.className = "console-tests-action-info";

                const label = document.createElement("span");
                label.className = "console-tests-action-label";
                label.textContent = action.label;
                info.appendChild(label);

                if (action.description) {
                    const detail = document.createElement("span");
                    detail.className = "console-tests-action-description";
                    detail.textContent = action.description;
                    info.appendChild(detail);
                }

                const button = document.createElement("button");
                button.type = "button";
                button.className = "console-tests-run";
                button.textContent = "RUN";
                button.addEventListener("click", () => {
                    this.runAction(suite, action, button, status);
                });

                row.append(info, button);
                section.appendChild(row);
            }

            root.appendChild(section);
        }

        this.panel.replaceChildren(root);
    }
}
